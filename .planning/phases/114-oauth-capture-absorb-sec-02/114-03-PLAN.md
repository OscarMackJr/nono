```rust
pub(crate) fn host_port_matches(pattern: &str, target: &str) -> bool {
    if pattern == target { return true; }
    if !pattern.starts_with("*.") { return false; }
    let Some((pattern_host, pattern_port)) = pattern.rsplit_once(':') else { return false; };
    let Some((target_host, target_port)) = target.rsplit_once(':') else { return false; };
    if pattern_port != target_port { return false; }
    let Some(suffix) = pattern_host.strip_prefix("*.") else { return false; };
    target_host.strip_suffix(suffix).is_some_and(|prefix| prefix.ends_with('.') && prefix.len() > 1)
}
```
