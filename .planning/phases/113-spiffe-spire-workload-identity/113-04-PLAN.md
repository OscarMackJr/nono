```rust
pub async fn load(routes: &[RouteConfig]) -> Result<Self> {
    let mut loaded = HashMap::new();
    for route in routes {
        let normalized_prefix = route.prefix.trim_matches('/').to_string();
        // ... existing endpoint_rules / endpoint_policy compile calls, unchanged ...
        // ... existing tls_ca -> (tls_connector, tls_client_config, tls_config_key) match, unchanged ...
        let upstream_host_port = extract_host_port(&route.upstream)/* unchanged */;

        // NEW: connect a SPIFFE JWT source when route.spiffe is declared. This is the ONLY
        // place a live Workload API connect happens for the plain-injection flow (the
        // client_assertion flow's connect lives in credential.rs::CredentialStore::load,
        // Plan 113-03 — the two are deliberately separate call sites for two separate configs).
        let (managed_auth, declares_spiffe) = match &route.spiffe {
            Some(nono_proxy::config::SpiffeAuthConfig::Jwt {
                workload_api_socket, audience, inject_header, credential_format, svid_hint,
            }) => {
                let source = crate::spiffe::SpiffeJwtSource::connect(
                    workload_api_socket, audience.clone(), inject_header.clone(),
                    credential_format.clone(), svid_hint.as_deref(),
                ).await.map_err(|e| ProxyError::Config(format!(
                    "route '{}': SPIFFE JWT source failed to connect: {e}", normalized_prefix
                )))?; // D-04/D-03: fails closed — the WHOLE proxy startup aborts, not a per-route skip.
                (Some(std::sync::Arc::new(crate::auth::ManagedUpstreamAuth::SpiffeJwt(std::sync::Arc::new(source)))), true)
            }
            None => (None, false),
        };

        loaded.insert(normalized_prefix, LoadedRoute {
            upstream: route.upstream.clone(),
            upstream_host_port: Some(upstream_host_port),
            endpoint_rules, endpoint_policy, tls_connector, tls_client_config, tls_config_key,
            managed_auth, declares_spiffe,
        });
    }
    Ok(Self { routes: loaded })
}
```

     unauthenticated" per c831dade's own commit message, and this is the literal RouteStore::load
     fail path D-04's text describes ("returns ProxyError::Config on failure"). The
     oauth2.client_assertion per-route-skip in credential.rs is this fork's OWN choice (matching its
     existing aws_auth/credential_key skip idiom for OTHER credential types) since upstream's

     declines to build (OD-1) — there is no upstream precedent to diverge from for that specific
     path. Do not conflate the two; both are intentional, not inconsistent. -->
